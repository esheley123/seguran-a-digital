import re import secrets import string def verificar_forca_senha(senha: str) -> dict: """ Analisa a força de uma senha com base em critérios de segurança digital. """ comprimento = len(senha) tem_maiuscula = bool(re.search(r'[A-Z]', senha)) tem_minuscula = bool(re.search(r'[a-z]', senha)) tem_numero = bool(re.search(r'[0-9]', senha)) tem_especial = bool(re.search(r'[^A-Za-z0-9]', senha)) # Contagem de critérios atendidos criterios_atendidos = sum([ comprimento >= 12, tem_maiuscula, tem_minuscula, tem_numero, tem_especial ]) # Classificação if comprimento < 8: classificacao = "Muito Fraca (Muito curta)" elif criterios_atendidos == 5: classificacao = "Forte" elif criterios_atendidos >= 3: classificacao = "Moderada" else: classificacao = "Fraca" return { "comprimento": comprimento, "maiuscula": tem_maiuscula, "minuscula": tem_minuscula, "numero": tem_numero, "especial": tem_especial, "classificacao": classificacao } def gerar_senha_segura(tamanho: int = 16) -> str: """ Gera uma senha segura utilizando a biblioteca secrets (criptograficamente segura). """ if tamanho < 12: tamanho = 12 # Força o mínimo recomendado por especialistas caracteres = string.ascii_letters + string.digits + string.punctuation senha = "".join(secrets.choice(caracteres) for _ in range(tamanho)) return senha def exibir_menu(): print("\n" + "="*40) print(" FERRAMENTA DE SEGURANÇA DIGITAL ") print("="*40) print("[1] Verificar Força de uma Senha") print("[2] Gerar uma Senha Segura") print("[3] Sair") print("="*40) def main(): while True: exibir_menu() opcao = input("Escolha uma opção (1-3): ").strip() if opcao == "1": senha = input("\nDigite a senha para teste: ") resultado = verificar_forca_senha(senha) print(f"\n--- Resultado da Análise ---") print(f"Classificação: {resultado['classificacao']}") print(f"Comprimento: {resultado['comprimento']} caracteres") print(f"Possui Letra Maiúscula: {'Sim' if resultado['maiuscula'] else 'Não'}") print(f"Possui Letra Minúscula: {'Sim' if resultado['minuscula'] else 'Não'}") print(f"Possui Número: {'Sim' if resultado['numero'] else 'Não'}") print(f"Possui Caractere Especial: {'Sim' if resultado['especial'] else 'Não'}") elif opcao == "2": try: tamanho = int(input("\nDigite o tamanho desejado da senha (mínimo 12): ")) except ValueError: tamanho = 16 senha_gerada = gerar_senha_segura(tamanho) print(f"\nSua senha segura gerada é: {senha_gerada}") elif opcao == "3": print("\nEncerrando o programa de segurança.
